
1. users:

- id (UUID, Primary Key)
- username (VARCHAR(50), Unique, Not Null)
- email (VARCHAR(100), Unique, Not Null)
- phone (VARCHAR(20), Not Null)
- password_hash (VARCHAR(255), Not Null)
- role (ENUM: 'admin', 'owner', 'penjaga', 'tenant', Not Null)
- full_name (VARCHAR(100), Not Null)
- created_at (TIMESTAMP, Default: CURRENT_TIMESTAMP)
- updated_at (TIMESTAMP, Default: CURRENT_TIMESTAMP on update)
- last_login (TIMESTAMP, Nullable)
- is_active (BOOLEAN, Default: true)
- profile_image_url (VARCHAR(255), Nullable)

2. properties:

- id (UUID, Primary Key)
- name (VARCHAR(100), Not Null)
- address (TEXT, Not Null)
- city (VARCHAR(50), Not Null)
- district (VARCHAR(50), Not Null)
- postal_code (VARCHAR(10), Not Null)
- contact_phone (VARCHAR(20), Not Null)
- property_manager_id (UUID, Foreign Key → users.id, Not Null)
- property_type (ENUM: 'male', 'female', 'mixed', Not Null)
- total_floors (INTEGER, Default: 0)
- total_rooms (INTEGER, Default: 0)
- amenities (JSON, Not Null) -- {wifi: true, ac: true, hot_water: true, ...}
- rules (TEXT, Nullable)
- status (ENUM: 'active', 'inactive', 'maintenance', Default: 'active')
- created_at (TIMESTAMP, Default: CURRENT_TIMESTAMP)
- updated_at (TIMESTAMP, Default: CURRENT_TIMESTAMP on update)
- floor_plan_images (JSON, Nullable) -- Array of image URLs per floor
- property_photos (JSON, Nullable) -- Array of property photo URLs

3. rooms:

- id (UUID, Primary Key)
- property_id (UUID, Foreign Key → properties.id, Not Null)
- room_number (VARCHAR(20), Not Null)
- floor (INTEGER, Not Null)
- room_type (ENUM: 'standard', 'deluxe', 'premium', Not Null)
- size_sqm (DECIMAL(5,2), Nullable)
- occupancy_type (ENUM: 'single', 'double', Default: 'single')
- base_monthly_rent (DECIMAL(12,2), Not Null)
- additional_person_fee (DECIMAL(12,2), Default: 0) -- For shared rooms
- amenities (JSON, Not Null) -- {ac: true, private_bathroom: true, ...}
- status (ENUM: 'available', 'occupied', 'reserved', 'maintenance', Default: 'available')
- ac_unit_id (VARCHAR(50), Nullable) -- For AC maintenance tracking
- coordinates_x (DECIMAL(10,6), Nullable) -- For floor plan positioning
- coordinates_y (DECIMAL(10,6), Nullable)
- created_at (TIMESTAMP, Default: CURRENT_TIMESTAMP)
- updated_at (TIMESTAMP, Default: CURRENT_TIMESTAMP on update)

4. room_media

- id (UUID, Primary Key)
- room_id (UUID, Foreign Key → rooms.id, Not Null)

- is_template (BOOLEAN, Default: false) -- True if used as group template

- Index: (room_id, display_order)
